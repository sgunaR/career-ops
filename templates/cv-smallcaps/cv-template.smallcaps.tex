% Custom commands
\newcommand{\resumeItem}[1]{
  \item\small{#1}
}

% Top-level entry row: bold left, plain right | italic left, italic right
\newcommand{\resumeSubheading}[4]{
  \vspace{2pt}\item
    \begin{tabular*}{0.97\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & #2 \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{0.5pt}
}

\renewcommand\labelitemii{$\circ$}

\newcommand{\resumeSubItem}[1]{\resumeItem{#1}}

% Top-level list: solid bullet
\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.12in, label=\textbullet, itemsep=1pt, topsep=1pt, parsep=0pt, partopsep=0pt]}

\newcommand{\resumeItemListStart}{\begin{itemize}[leftmargin=0.18in, label=$\circ$, itemsep=0.5pt, topsep=0.5pt, parsep=0pt, partopsep=0pt]}

\newcommand{\plainListStart}{\begin{itemize}[leftmargin=0.12in, label=\textbullet, itemsep=1pt, topsep=1pt, parsep=0pt, partopsep=0pt]}
\newcommand{\plainListEnd}{\end{itemize}\vspace{0.5pt}}
